223  class KeyCmp {
224  public:
225  static std::size_t BitLength(const InetRouteFlowMgmtKey *rt) {
226  return (((sizeof(rt->vrf_id_) + sizeof(rt->type_)) << 3) +
227  rt->plen_);
228  }

230  static char ByteValue(const InetRouteFlowMgmtKey *rt, std::size_t idx) {
231  const char *ch;
232  std::size_t i = idx;
233  if (i < sizeof(rt->vrf_id_)) {
234  ch = (const char *)&rt->vrf_id_;
235  return ch[sizeof(rt->vrf_id_) - i - 1];
236  }
237  i -= sizeof(rt->vrf_id_);
238  if (i < sizeof(rt->type_)) {
239  ch = (const char *)&rt->type_;
240  return ch[sizeof(rt->type_) - i - 1];
241  }
242  i -= sizeof(rt->type_);
243  if (rt->type_ == INET4) {
244  return rt->ip_.to_v4().to_bytes()[i];
245  } else if (rt->type_ == INET6) {
246  return rt->ip_.to_v6().to_bytes()[i];
247  } else {
248  assert(0);
249  return 0;
250  }
251  }
252  };
