523 int EvpnRouteEntry::CompareTo(const Route &rhs) const {
524  const EvpnRouteEntry &a = static_cast<const EvpnRouteEntry &>(rhs);
525 
526  if (ethernet_tag_ < a.ethernet_tag_)
527  return -1;
528 
529  if (ethernet_tag_ > a.ethernet_tag_)
530  return 1;
531 
532  int cmp = mac_.CompareTo(a.mac_);
533  if (cmp != 0)
534  return cmp;
535 
536  if (prefix_address_ < a.prefix_address_)
537  return -1;
538 
539  if (prefix_address_ > a.prefix_address_)
540  return 1;
541 
542  if (prefix_length() < a.prefix_length())
543  return -1;
544 
545  if (prefix_length() > a.prefix_length())
546  return 1;
547 
548  return 0;
549 }

582 //Notify L2 route corresponding to MAC in evpn route.
583 //On addition of evpn routes, update bridge route using mac of evpn and inet
584 //route using ip of evpn.
585 //NH of bridge route is same as of EVPN as bridge rt is programmed in kernel.
586 //NH of Inet route will be of subnet route to which it belongs. This makes sure
587 //that n absence of any directly installed Inet route for this IP packets are
588 //forwarded as per the subnet route decision.
589 void EvpnRouteEntry::UpdateDerivedRoutes(AgentRouteTable *table,
590  const AgentPath *path,
591  bool active_path_changed) {
592  //As active path is picked from route, any modification in non-active
593  //path need not rebake agent route.
594  //Path is NULL when resync is issued on route, hence no need to check flag
595  if ((path != NULL) && !active_path_changed)
596  return;
597 
598 
599  if (publish_to_bridge_route_table()) {
600  BridgeAgentRouteTable *bridge_table = NULL;
601  bridge_table = static_cast<BridgeAgentRouteTable *>
602  (table->vrf_entry()->GetBridgeRouteTable());
603  if (bridge_table)
604  bridge_table->AddBridgeRoute(this);
605  }
606  if (publish_to_inet_route_table()) {
607  InetUnicastAgentRouteTable *inet_table =
608  table->vrf_entry()->GetInetUnicastRouteTable(prefix_address());
609  if (inet_table)
610  inet_table->AddEvpnRoute(this);
611  }
612 }
