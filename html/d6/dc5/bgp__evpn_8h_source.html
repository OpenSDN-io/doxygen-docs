761 inline void intrusive_ptr_add_ref(EvpnState *evpn_state) {
762  evpn_state->refcount_.fetch_add(1);
763 }
764 
769 inline void intrusive_ptr_release(EvpnState *evpn_state) {
770  int prev = evpn_state->refcount_.fetch_sub(1);
771  if (prev > 1)
772  return;
773  if (evpn_state->states()) {
774  EvpnState::StatesMap::iterator iter =
775  evpn_state->states()->find(evpn_state->sg());
776  if (iter != evpn_state->states()->end()) {
777  assert(iter->second == evpn_state);
778  evpn_state->states()->erase(iter);
779 
780  // Attempt project manager deletion as it could be held up due to
781  // this map being non-empty so far..
782  if (evpn_state->manager()->deleter()->IsDeleted())
783  evpn_state->manager()->deleter()->RetryDelete();
784  }
785  }
786  delete evpn_state;
787 }
