144 struct Dhcpv6Options {
145  void WriteData(uint16_t c, uint16_t l, const void *d, uint16_t *optlen) {
146  code = htons(c);
147  len = htons(l);
148  memcpy(data, (uint8_t *)d, l);
149  *optlen += 4 + l;
150  }
151  void AppendData(uint16_t l, const void *d, uint16_t *optlen) {
152  uint16_t curr_len = ntohs(len);
153  memcpy(data + curr_len, (uint8_t *)d, l);
154  len = htons(curr_len + l);
155  *optlen += l;
156  }
157  void AddLen(uint16_t l) {
158  len = htons(ntohs(len) + l);
159  }
160  uint16_t GetLen() const { return ntohs(len); }
161  Dhcpv6Options *GetNextOptionPtr() {
162  uint8_t *next = reinterpret_cast<uint8_t *>(this);
163  return reinterpret_cast<Dhcpv6Options *>(next + ntohs(len) + 4);
164  }
165 
166  uint16_t code;
167  uint16_t len;
168  uint8_t data[0];
169 };

190 struct Dhcpv6IaAddr {
191  Dhcpv6IaAddr(Dhcpv6IaAddr *ptr) {
192  if (ptr) {
193  memcpy(address, ptr->address, 16);
194  preferred_lifetime = ptr->preferred_lifetime;
195  valid_lifetime = ptr->valid_lifetime;
196  } else {
197  memset(address, 0, 16);
198  preferred_lifetime = 0;
199  valid_lifetime = 0;
200  }
201  }
202  Dhcpv6IaAddr(uint8_t *addr, uint32_t pl, uint32_t vl) {
203  memcpy(address, addr, 16);
204  preferred_lifetime = htonl(pl);
205  valid_lifetime = htonl(vl);
206  }
207 
208  uint8_t address[16];
209  uint32_t preferred_lifetime;
