52 bool PhysicalDeviceVn::Copy(PhysicalDeviceVnTable *table,
53  const PhysicalDeviceVnData *data) {
54  bool ret = false;
55 
56  PhysicalDevice *dev =
57  table->agent()->physical_device_table()->Find(device_uuid_);
58  if (dev != device_.get()) {
59  device_.reset(dev);
60  ret = true;
61  }
62 
63  if (dev && (dev->ip() != tor_ip_)) {
64  tor_ip_ = dev->ip();
65  ret = true;
66  }
67 
68  if (dev && (dev->name() != device_display_name_)) {
69  device_display_name_ = dev->name();
70  ret = true;
71  }
72 
73  VnEntry *vn = table->agent()->vn_table()->Find(vn_uuid_);
74  if (vn != vn_.get()) {
75  vn_.reset(vn);
76  ret = true;
77  }
78 
79  int vxlan_id = vn ? vn->GetVxLanId() : 0;
80  if (vxlan_id != vxlan_id_) {
81  vxlan_id_ = vxlan_id;
82  ret = true;
83  }
84 
85  return ret;
86 }

192 bool PhysicalDeviceVnTable::AddConfigEntry(const boost::uuids::uuid &vmi,
193  const boost::uuids::uuid &dev,
194  const boost::uuids::uuid &vn) {
195  using boost::uuids::nil_uuid;
196 
197  // Sanity checks. Needed since VMInterface is not checking for nil_uuid

206 bool PhysicalDeviceVnTable::DeleteConfigEntry(const boost::uuids::uuid &vmi,
207  const boost::uuids::uuid &dev,
208  const boost::uuids::uuid &vn) {
209  using boost::uuids::nil_uuid;
210 
211  // Sanity checks. Needed since VMInterface is not checking for nil_uuid
212  if (vmi == nil_uuid() || dev == nil_uuid() || vn == nil_uuid())
213  return false;
214 
215  config_tree_.erase(PhysicalDeviceVnToVmi(dev, vn, vmi));
216  // Dont delete physical-device-vn entry if there are more entries in
217  // config-tree with given dev and vn
218  ConfigTree::iterator it =
219  config_tree_.upper_bound(PhysicalDeviceVnToVmi(dev, vn, nil_uuid()));
220  bool del_entry = false;
221  if (it == config_tree_.end())
222  del_entry = true;
223  else if (it->dev_ != dev)
224  del_entry = true;
225  else if (it->vn_ != vn)
226  del_entry = true;
227 
228  if (del_entry) {
229  agent()->config_manager()->DelPhysicalDeviceVn(dev, vn);
230  DBRequest req(DBRequest::DB_ENTRY_DELETE);
231  req.key.reset(new PhysicalDeviceVnKey(dev, vn));
232  Enqueue(&req);
233  }
234 
235  return del_entry;
236 }
